% ===================================================================
\section{TC-012 -- \texttt{CSV}}
% ===================================================================
\label{sec:tc012}

Requirement reference: REQ-012.

% -------------------------------------------------------------------
\subsection{TC-012-A: A not-existant file cannot be parsed}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-012-1

\medskip
\given{}
\when{Calling \texttt{pacemaker::io::CSV::parse("")}}
\then{An \texttt{std::invalid\_argument} exception is raised.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-012-B: A directory cannot be parsed}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-012-2

\medskip
\given{The OS specific temporary folder.}
\when{Calling \texttt{pacemaker::io::CSV::parse} with the path to the temporary folder directory}
\then{An \texttt{std::invalid\_argument} exception is raised.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-012-C: The record delimiter is $\backslash$r, $\backslash$n, and/or $\backslash$r$\backslash$n and the fields are returned in order}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-012-3, AC-012-5

\medskip
\given{A file created in the temporary folder with the string literal "a$\backslash$rb$\backslash$nc$\backslash$r$\backslash$nd"}
\when{Calling \texttt{pacemaker::io::CSV::parse} with the path to the created file}
\then{The returned structure contains a vector of 4 vectors of string, where each first element are the letter "a", "b", "c", and "d"}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-012-D: The field delimiter is a comma and the fields are returned in order}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-012-4

\medskip
\given{A file created in the temporary folder with the string literal "a,b"}
\when{Calling \texttt{pacemaker::io::CSV::parse} with the path to the created file}
\then{The returned inner vector contains the strings "a" and "b"}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-012-E: Empty fields are valid fields}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-012-6

\medskip
\given{A file created in the temporary folder with the string literal ",,,"}
\when{Calling \texttt{pacemaker::io::CSV::parse} with the path to the created file}
\then{The returned inner vector contains four empty strings}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-012-F: The returned structure preserves the order of records and fields in the source file}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-012-7

\medskip
\given{A file created in the temporary folder with the string literal "0,1,2$\backslash$n3,4,5"}

% -------------------------------------------------------------------

\tcfield{Requirement:} AC-012-8

\medskip

\begin{itemize}
  \item \textit{(0x20)}

  \item \textit{(0x2D)}
  \item \textit{(0x7E)}
  \item \#\$\%\&'()*+
  \item -./\textit{0...9}:;<=>?@\textit{A...Z}[$\backslash$]$\hat{}$\_\textit{a...z}\{ \}~

  \begin{itemize}
  \item \{\{\textit{(0x20)}\}\}

  \item \{\{\textit{(0x23)}\}\}
  \item \{\{\textit{(0x2B)}\}\}
  \item \{\{\textit{(0x2D)}\}\}

  \item \{\{"-./\textit{0...9}:;<=>?@\textit{A...Z}[$\backslash$]$\hat{}$\_\textit{a...z}\{ \}~"\}\}
  \item \{\{\textit{leading spaces are preserved}\}\}
  \item \{\{"hello world"\}\}
\end{itemize}
}
\end{tcbox}
