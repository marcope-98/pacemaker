% ===================================================================
\section{TC-009 -- \texttt{Session}}
% ===================================================================
\label{sec:tc009}

Requirement reference: REQ-009.

% -------------------------------------------------------------------
\subsection{TC-009-A: Session::connect() succeeds with INCA running}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[integration]}\\
\tcfield{Requirement:} AC-009-1, AC-009-3, AC-009-5, AC-009-6

\medskip
\given{INCA is installed, running, has an experiment open with at least one device online, and COM has been initialised via \texttt{CoInitializeEx}.}
\when{\texttt{Session::connect()} is called.}
\then{a valid \texttt{Session} is returned.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-009-B: connect() throws when INCA is not installed}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[integration]}\\
\tcfield{Requirement:} AC-009-2

\medskip
\given{INCA is \emph{not} installed on the test host (or the CLSID is unregistered).}
\when{\texttt{Session::connect()} is called.}
\then{\texttt{std::runtime\_error} is thrown with a message containing the word \texttt{"INCA"}.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-009-C: connect() throws when no device is online}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-009-4

\medskip
\given{INCA is running, an experiment is open, but \emph{no} device is configured or online.}
\when{\texttt{Session::connect()} is called.}
\then{\texttt{std::runtime\_error} is thrown with a message indicating no device is available.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-009-D: Session is non-copyable and non-movable}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]} (compile-time check)\\
\tcfield{Requirement:} AC-009-7

\medskip
\given{the \texttt{Session} type definition.}
\when{copy and move traits are evaluated at compile time.}
\then{\texttt{std::is\_copy\_constructible\_v<Session>} is \texttt{false}}
\andthen{\texttt{std::is\_copy\_assignable\_v<Session>} is \texttt{false}}
\andthen{\texttt{std::is\_move\_constructible\_v<Session>} is \texttt{true}}
\andthen{\texttt{std::is\_move\_assignable\_v<Session>} is \texttt{true}.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-009-F: add\_param is idempotent}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-009-9, AC-009-10

\medskip
\given{a \texttt{Session} constructed with mock proxies.}
\andgiven{a mock that returns a distinct \texttt{IDispatch} object each time \texttt{GetCalibrationValueInDevice} is called.}
\when{\texttt{add\_param("speed")} is called twice.}
\then{\texttt{GetCalibrationValueInDevice} is called exactly once (the second call is no-op).}
\end{tcbox}
