% ===================================================================
\section{TC-003 -- \texttt{IncaProxy}}
% ===================================================================
\label{sec:tc003}

Requirement reference: REQ-003.

\textit{Note: tests in this section require either a running INCA instance
(\texttt{[integration]}) or a mock COM server (\texttt{[unit]}).}

% -------------------------------------------------------------------
\subsection{TC-003-A: Construction narrows IDispatch successfully}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-003-1

\medskip
\given{a mock \texttt{IDispatch} that implements the \texttt{Inca\_Dispatch} interface UUID.}
\when{\texttt{IncaProxy} is constructed with that pointer.}
\then{no exception is thrown and the proxy is valid.}

\bigskip
\begin{tabular}{@{}L{0.5cm}L{5cm}L{7cm}@{}}
\toprule
\textbf{\#} & \textbf{Step} & \textbf{Expected result} \\
\midrule
1 & Create mock implementing \texttt{Inca\_Dispatch} UUID. & Mock ready. \\
2 & Construct \texttt{IncaProxy proxy\{std::move(idispatch)\}}. & No exception; proxy alive. \\
\bottomrule
\end{tabular}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-003-B: Destructor calls DisconnectFromTool}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-003-2

\medskip
\given{a mock \texttt{Inca\_Dispatch} that records calls to \texttt{DisconnectFromTool}.}
\when{the owning \texttt{IncaProxy} is destroyed.}
\then{\texttt{DisconnectFromTool()} has been called exactly once.}

\bigskip
\begin{tabular}{@{}L{0.5cm}L{5cm}L{7cm}@{}}
\toprule
\textbf{\#} & \textbf{Step} & \textbf{Expected result} \\
\midrule
1 & Create mock; wrap in \texttt{IncaProxy}. & Call counter = 0. \\
2 & Destroy \texttt{IncaProxy} (end of scope). & \texttt{DisconnectFromTool()} counter = 1. \\
\bottomrule
\end{tabular}
\end{tcbox}
