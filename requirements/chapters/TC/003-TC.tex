% -------------------------------------------------------------------
\subsection{TC-003-A: GetOpenedExperiment returns valid IncaOnlineExperimentProxy}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-003-3

\medskip
\given{a mock class \texttt{MockCOM} whose \texttt{GetOpenedExperiment()} returns a non-null \texttt{IDispatch*}.}
\when{\texttt{proxy->GetOpenedExperiment()} is called.}
\then{no exceptions are thrown and the returned \texttt{IncaOnlineExperimentProxy} is non-null.}
\end{tcbox}
