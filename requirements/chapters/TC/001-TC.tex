% ===================================================================
\section{TC-001 -- \texttt{unique\_com\_ptr}}
% ===================================================================
\label{sec:tc001}

Requirement reference: REQ-001.

% -------------------------------------------------------------------
\subsection{TC-001-A: Release is called on destruction}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-001-1

\medskip
\given{a mock COM object with a mocked \texttt{Release()} method.}
\when{a \texttt{unique\_com\_ptr} owning that object goes out of scope.}
\then{\texttt{Release()} has been called exactly once.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-001-B: No Release on null pointer}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-001-2

\medskip
\given{ a mock COM object with a mocked \texttt{Release()} method}
\andgiven{ a default-constructed (nullptr) \texttt{unique\_com\_ptr<...>}} 
\when{the wrapper is destroyed.}
\then{\texttt{Release()} has never been called.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-001-C: Move transfers ownership without extra Release}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-001-3

\medskip
\given{a \texttt{unique\_com\_ptr} owning a mock COM object.}
\when{the pointer is moved into a second \texttt{unique\_com\_ptr} and both go out of scope.}
\then{\texttt{Release()} is called exactly once and the moved-from wrapper is null.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-001-D: Copy construction is deleted}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]} (compile-time check)\\
\tcfield{Requirement:} AC-001-4

\medskip
\given{a \texttt{unique\_com\_ptr<T>} type.}
\when{a copy-construction expression is compiled.}
\then{the compilation fails with a deleted-function error.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-001-E: \texttt{get()} does not affect ownership.}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-001-5

\medskip
\given{a \texttt{unique\_com\_ptr<T>} owning a mock COM object.}
\when{\texttt{get()} is called inside a scope.}
\then{the returned pointer is the same as the address of the COM object and \texttt{Release()} is called once.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-001-F: The type is compatible with \texttt{std::unique\_ptr}.}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-001-6

\medskip
\given{a \texttt{unique\_com\_ptr<T>} type.}
\when{checking if it is an alias of \texttt{unique\_ptr}.}
\then{the result is true.}
\end{tcbox}
