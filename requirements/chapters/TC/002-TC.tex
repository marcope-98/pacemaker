% ===================================================================
\section{TC-002 -- \texttt{COMProxy<T>}}
% ===================================================================
\label{sec:tc002}

Requirement reference: REQ-002.

% -------------------------------------------------------------------
\subsection{TC-002-A: Copy is deleted, move is available}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]} (compile-time check)\\
\tcfield{Requirement:} AC-002-1, AC-002-2

\medskip
\given{a mock class \texttt{MockCOM} that inherits from IDispatch and mocks its methods.}
\andgiven{a \texttt{COMProxy<MockCOM>} object.}
\when{copy and move constructibility are queried at compile time.}
\then{\texttt{std::is\_copy\_constructible\_v<ConcreteProxy>} is \texttt{false}}
\andthen{\texttt{std::is\_copy\_assignable\_v<ConcreteProxy>} is \texttt{false}}
\andthen{\texttt{std::is\_move\_constructible\_v<ConcreteProxy>} is \texttt{true}}
\andthen{\texttt{std::is\_move\_assignable\_v<ConcreteProxy>} is \texttt{true}.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-002-B: Underlying type methods are accessible via \texttt{operator->} and \texttt{get()}}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-002-6

\medskip
\given{a mock class \texttt{MockCOM} that inherits from IDispatch and mocks its methods.}
\andgiven{a generic method \texttt{MockCOM::func()}.}
\andgiven{a \texttt{COMProxy<MockCOM>} object.}
\when{Calling \texttt{func()} via the arrow operator and the \texttt{get()} method.}
\then{The method \texttt{func()} gets called twice and does not throw exceptions.}
\end{tcbox}
