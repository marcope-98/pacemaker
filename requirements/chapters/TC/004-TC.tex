% ===================================================================
\section{TC-004 -- \texttt{IncaOnlineExperimentProxy}}
% ===================================================================
\label{sec:tc004}

Requirement reference: REQ-004.

% -------------------------------------------------------------------
\subsection{TC-004-A: GetAllDevices returns correct count}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-004-2

\medskip
\given{a mock \texttt{IncaOnlineExperiment\_Dispatch} whose \texttt{GetAllDevices()} returns a \texttt{SAFEARRAY} containing two \texttt{VT\_DISPATCH} elements.}
\when{\texttt{proxy->GetAllDevices()} is called.}
\then{the returned vector has exactly two elements, each of type non-null ExperimentDeviceProxy.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-004-B: GetAllDevices throws on wrong SAFEARRAY type}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-004-3

\medskip
\given{a mock returning a \texttt{\_variant\_t} of type \texttt{VT\_I4} (wrong type) from \texttt{GetAllDevices()}.}
\when{\texttt{proxy->GetAllDevices()} is called.}
\then{\texttt{std::runtime\_error} is thrown.}
\end{tcbox}

% -------------------------------------------------------------------
\subsection{TC-004-C: GetCalibrationValueInDevice returns CalibrationScalarDataProxy}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-004-7

\medskip
\given{a mock \texttt{IncaOnlineExperiment\_Dispatch} whose \texttt{GetCalibrationValueInDevice()} returns a valid \texttt{IDispatch*}.}
\when{\texttt{auto value = proxy->GetCalibrationValueInDevice("unknownParam", device)} is called.}
\then{\texttt{value} has the same type as \texttt{CalibrationScalarDataProxy}}
\end{tcbox}


% -------------------------------------------------------------------
\subsection{TC-004-D: GetCalibrationValueInDevice throws when COM returns null}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-004-8

\medskip
\given{a mock \texttt{IncaOnlineExperiment\_Dispatch} whose \texttt{GetCalibrationValueInDevice()} returns a nullptr \texttt{IDispatch*}.}
\when{\texttt{proxy->GetCalibrationValueInDevice("unknownParam", device)} is called.}
\then{\texttt{std::runtime\_error} is thrown with a message containing the parameter name.}
\end{tcbox}
