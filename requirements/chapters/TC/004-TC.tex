% -------------------------------------------------------------------
\subsection{TC-004-A: GetAllDevices returns correct count}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-004-2

\medskip
\given{a mock \texttt{IncaOnlineExperiment\_Dispatch} whose \texttt{GetAllDevices()} returns a \texttt{SAFEARRAY} containing two \texttt{VT\_DISPATCH} elements.}
\when{\texttt{proxy.GetAllDevices()} is called.}
\then{the returned vector has exactly two elements, each non-null.}

\bigskip
\begin{tabular}{@{}L{0.5cm}L{5cm}L{7cm}@{}}
\toprule
\textbf{\#} & \textbf{Step} & \textbf{Expected result} \\
\midrule
1 & Build a \texttt{SAFEARRAY} of 2 \texttt{VARIANT}s, each \texttt{VT\_DISPATCH} with mock \texttt{IDispatch*}. & Array ready. \\
2 & Configure mock to return the array from \texttt{GetAllDevices()}. & Mock ready. \\
3 & Call \texttt{proxy.GetAllDevices()}. & No exception thrown. \\
4 & Assert \texttt{result.size() == 2}. & Passes. \\
5 & Assert \texttt{result[0] != nullptr \&\& result[1] != nullptr}. & Passes. \\
\bottomrule
\end{tabular}
\end{tcbox}
