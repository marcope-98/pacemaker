% ===================================================================
\section{TC-008 -- \texttt{Experiment}}
% ===================================================================
\label{sec:tc008}

Requirement reference: REQ-008.

For this section all COM proxies are replaced by lightweight mock wrappers that expose
the same interface and record calls.

% -------------------------------------------------------------------
\subsection{TC-008-A: add\_param is idempotent}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-008-3

\medskip
\given{an \texttt{Experiment} constructed with mock proxies, and a mock that returns a distinct \texttt{IDispatch} object each time \texttt{GetCalibrationValueInDevice} is called.}
\when{\texttt{add\_param("speed")} is called twice.}
\then{\texttt{GetCalibrationValueInDevice} is called exactly once (the second call is a no-op).}

\bigskip
\begin{tabular}{@{}L{0.5cm}L{5cm}L{7cm}@{}}
\toprule
\textbf{\#} & \textbf{Step} & \textbf{Expected result} \\
\midrule
1 & Call \texttt{experiment.add\_param("speed")}. & COM call count = 1; map size = 1. \\
2 & Call \texttt{experiment.add\_param("speed")} again. & COM call count still = 1; map size still = 1. \\
\bottomrule
\end{tabular}
\end{tcbox}

\subsection{TC-008-E: stop\_recording sleeps at least 40\,ms before delegating}
% -------------------------------------------------------------------

\begin{tcbox}
\tcfield{Tags:} \texttt{[unit]}\\
\tcfield{Requirement:} AC-008-2, AC-008-7

\medskip
\given{an \texttt{Experiment} backed by mocks, and a high-resolution timer.}
\when{\texttt{stop\_recording("")} is called.}
\then{the elapsed wall-clock time from entry to the first COM call is at least 40\,ms; \texttt{StopRecordingAndSave()} is called before \texttt{StopMeasurement()}.}

\bigskip
\begin{tabular}{@{}L{0.5cm}L{5cm}L{7cm}@{}}
\toprule
\textbf{\#} & \textbf{Step} & \textbf{Expected result} \\
\midrule
1 & Record \texttt{t\_start} using \texttt{std::chrono::steady\_clock::now()}. & Baseline captured. \\
2 & Call \texttt{experiment.stop\_recording("")}. & Method blocks for $\geq$ 40\,ms. \\
3 & Record \texttt{t\_end} after method returns. & $t\_end - t\_start \geq 40$\,ms. \\
4 & Assert mock call order: \texttt{StopRecordingAndSave} before \texttt{StopMeasurement}. & Order correct. \\
5 & Assert \texttt{reset()} was invoked (map is empty). & Passes. \\
\bottomrule
\end{tabular}
\end{tcbox}
