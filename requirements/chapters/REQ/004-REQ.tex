% ===================================================================
\section{REQ-004 -- \texttt{IncaOnlineExperimentProxy}}
% ===================================================================
\label{sec:incaonlineexperimentproxy}

\reqfield{Title:} Proxy for the INCA online experiment COM object \status{implemented}

\medskip
\reqfield{Intent:}
The user needs to start and stop measurement recordings, enumerate connected ECU devices, and retrieve calibration data items from a specific device, all through a safe C++ API that hides COM SAFEARRAY marshalling and BSTR string handling.

\medskip
\reqfield{Acceptance criteria:}
\reqdef{\textbf{AC-004-2}}{\texttt{GetAllDevices()} shall return a \texttt{std::vector} of \texttt{ExperimentDeviceProxy}, one element per device reported by INCA.}
\reqdef{\textbf{AC-004-3}}{\texttt{GetAllDevices()} shall throw \texttt{std::runtime\_error} if the returned SAFEARRAY is not of type \texttt{VT\_ARRAY | VT\_VARIANT} or if any element is not \texttt{VT\_DISPATCH}.}
\reqdef{\textbf{AC-004-7}}{\texttt{GetCalibrationValueInDevice(name, device)} shall return a \texttt{CalibrationScalarDataProxy}.}
\reqdef{\textbf{AC-004-8}}{\texttt{GetCalibrationValueInDevice(name, device)} shall throw \texttt{std::runtime\_error} if the underlying COM call returns a null pointer.}
\reqdef{\textbf{AC-004-10}}{The class shall respect all the requirements of \texttt{COMProxy}.}

\medskip
