% ===================================================================
\section{REQ-003 -- \texttt{IncaProxy}}
% ===================================================================
\label{sec:incaproxy}

\reqfield{Title:} Proxy for the top-level INCA COM object \status{implemented}

\medskip
\reqfield{Intent:}
The user needs a C++ object that connects to a running INCA instance and retrieves the
currently open experiment and experiment-view COM objects.

\medskip
\reqfield{Acceptance criteria:}
\reqdef{\textbf{AC-003-3}}{\texttt{GetOpenedExperiment()} shall return an \texttt{IncaOnlineExperimentProxy} owning the experiment object.}
\reqdef{\textbf{AC-003-4}}{\texttt{GetOpenedExperimentView()} shall return an \texttt{IncaExperimentViewProxy} owning the experiment-view object.}
\reqdef{\textbf{AC-003-6}}{The class shall respect all the requirements of \texttt{COMProxy}.}

\medskip
\reqfield{Testing:}
See test document, section \textit{TC-003}.
