% ===================================================================
\section{REQ-003 -- \texttt{IncaProxy}}
% ===================================================================
\label{sec:incaproxy}

\reqfield{Title:} Proxy for the top-level INCA COM object \statusupd

\changenote{\\
AC-003-2 is rejected: \texttt{DisconnectFromTool()} shall no longer be called in the proxy destructor. \\
AC-003-3 and AC-003-4 updated: getters now shall return fully typed proxy objects rather than raw \\
\texttt{unique\_com\_ptr<IDispatch>}.}

\medskip
\reqfield{Intent:}
The user needs a C++ object that connects to a running INCA instance and retrieves the
currently open experiment and experiment-view COM objects.

\medskip
\reqfield{Acceptance criteria:}
\acrej{\textbf{AC-003-1}}{The constructor shall narrow the supplied \texttt{IDispatch} to \texttt{Inca\_Dispatch} via \texttt{query\_interface}; it shall throw if the narrowing fails.}
\acrej{\textbf{AC-003-2}}{The destructor shall call \texttt{DisconnectFromTool()} if the internal pointer is non-null.}
\acupd{\textbf{AC-003-3}}{\texttt{GetOpenedExperiment()} shall return an \texttt{IncaOnlineExperimentProxy} owning the experiment object.}
\acupd{\textbf{AC-003-4}}{\texttt{GetOpenedExperimentView()} shall return an \texttt{IncaExperimentViewProxy} owning the experiment-view object.}
\acobs{\textbf{AC-003-5}}{The class shall be move-only.}
\acnew{\textbf{AC-003-6}}{The class shall respect all the requirements of \texttt{COMProxy}.}

\medskip
\reqfield{Testing:}
See test document, section \textit{TC-003}.
