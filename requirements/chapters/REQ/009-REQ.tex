% ===================================================================
\section{REQ-009 -- \texttt{Session}}
% ===================================================================
\label{sec:session}

\reqfield{Title:} Top-level session factory and lifetime manager \status{implemented}

\medskip
\reqfield{Intent:}
The user needs a single entry point that connects to INCA, validates the environment, and exposes the complete measurement workflow, parameter management, recording control, and teardown, through one object.

\medskip
\reqfield{Acceptance criteria:}
\reqdef{\textbf{AC-009-1}}{\texttt{Session::connect()} shall call \texttt{CoCreateInstance} to initialize Inca COM object.}
\reqdef{\textbf{AC-009-2}}{If \texttt{CoCreateInstance} fails, \texttt{connect()} shall throw \texttt{std::runtime\_error} with a message indicating that INCA may not be installed.}
\reqdef{\textbf{AC-009-3}}{\texttt{connect()} shall retrieve the open experiment and experiment-view objects via \texttt{IncaProxy}.}
\reqdef{\textbf{AC-009-4}}{\texttt{connect()} shall enumerate devices; if none are found it shall throw \texttt{std::runtime\_error} with a message indicating no device is configured or online.}
\reqdef{\textbf{AC-009-5}}{On success, \texttt{connect()} shall return a \texttt{Session} constructed via its private default constructor.}
\reqdef{\textbf{AC-009-7}}{The class shall be non-copyable and non-movable.}
\reqdef{\textbf{AC-009-8}}{The destructor shall call \texttt{DisconnectFromTool()} on the \texttt{IncaProxy} if the proxy is non-null.}
\reqdef{\textbf{AC-009-9}}{\texttt{add\_param(name)} shall look up the named calibration scalar on the primary device and store the resulting \texttt{CalibrationScalarDataProxy} internally.}
\reqdef{\textbf{AC-009-10}}{Calling \texttt{add\_param} two or more times with the same name shall be a no-op.}
\reqdef{\textbf{AC-009-11}}{\texttt{set\_param(name, value)} shall write the value to the registered calibration scalar.}
\reqdef{\textbf{AC-009-12}}{Calling \texttt{set\_param(name, value)} with a \texttt{name} that was not registered via \texttt{add\_param}, shall return without taking further actions.}
\reqdef{\textbf{AC-009-13}}{\texttt{reset()} shall call \texttt{ResetValueToRP()} on every registered calibration proxy.}
\reqdef{\textbf{AC-009-14}}{\texttt{start\_recording()} shall delegate to \texttt{IncaOnlineExperimentProxy::StartRecording()} via \texttt{operator->}.}
\reqdef{\textbf{AC-009-15}}{\texttt{stop\_recording(filename)} shall: sleep for (40\,ms) before saving stopping the recording and resetting all calibration values to the Reference Page.}
\reqdef{\textbf{AC-009-16}}{It shall be possible to initialize \texttt{Session} by providing a lambda expression via \texttt{Session::connect(...)}.}

\medskip
\reqfield{Non-functional constraints:}
\begin{itemize}[leftmargin=*]
  \item COM must be initialised by the caller before invoking \texttt{connect()} (e.g.\ via \texttt{CoInitializeEx}).
  \item Platform: Windows only.
\end{itemize}
